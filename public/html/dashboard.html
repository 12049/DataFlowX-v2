const BASE_URL = 'https://blackwave-api.vercel.app/api/v1';

    const response = await axios.post(

  }
}

import requests, json

API_KEY = 'YOUR_API_KEY_HERE'

    headers = { 'Authorization': f'Bearer {API_KEY}', 'Content-Type': 'application/json' }

    response = requests.get(f'{BASE_URL}/download/youtube', headers=headers, params=params)
    if response.status_code == 200: return response.json()

// Java example using HttpClient
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.net.http.HttpRequest.BodyPublishers;
import java.net.http.HttpResponse.BodyHandlers;
import java.time.Duration;

public class BlackwaveApiExample {
    private static final String API_KEY = "YOUR_API_KEY_HERE";
    private static final String BASE_URL = "https://blackwave-api.vercel.app/api/v1";

    public static void main(String[] args) throws Exception {
        HttpClient client = HttpClient.newBuilder()
            .version(HttpClient.Version.HTTP_2)
            .connectTimeout(Duration.ofSeconds(10))
            .build();

        String jsonBody = "{\"text\":\"Hello from Blackwave API\",\"voice\":\"en-US-1\"}";

        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create(BASE_URL + "/tools/text-to-speech"))
            .header("Authorization", "Bearer " + API_KEY)
            .header("Content-Type", "application/json")
            .POST(BodyPublishers.ofString(jsonBody))
            .build();

        HttpResponse response = client.send(request, BodyHandlers.ofString());
        System.out.println("Status Code: " + response.statusCode());
        System.out.println("Response: " + response.body());
    }
}
